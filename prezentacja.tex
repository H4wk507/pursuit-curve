\section{Wprowadzenie}
% ============================================

\begin{frame}{Czym jest problem pościgu?}
\textbf{Problem:} Określenie trajektorii ścigającego, który dąży do przechwycenia poruszającego się celu.

\vspace{0.5cm}
\textbf{Zastosowania:}
\begin{itemize}
    \item Naprowadzanie rakiet i pocisków
    \item Roboty autonomiczne i drony
    \item Nawigacja morska i lotnicza
    \item AI w grach komputerowych
    \item Optymalizacja trajektorii
\end{itemize}

\vspace{0.5cm}
\textbf{Podejście:} Rozwiązywanie równań różniczkowych opisujących ruch ścigającego i celu.
\end{frame}

% ============================================
\section{Strategie pościgu}
% ============================================

\begin{frame}{Pościg bezpośredni (Direct Pursuit)}
Ścigający w każdej chwili celuje \textbf{bezpośrednio} w aktualną pozycję celu.

\vspace{0.5cm}
\textbf{Równanie ruchu:}
\begin{equation*}
\frac{d\vec{p}}{dt} = v_p \cdot \frac{\vec{t} - \vec{p}}{\|\vec{t} - \vec{p}\|}
\end{equation*}

gdzie:
\begin{itemize}
    \item $\vec{p}$ - pozycja ścigającego
    \item $\vec{t}$ - pozycja celu
    \item $v_p$ - prędkość ścigającego
\end{itemize}

\vspace{0.3cm}
\textbf{Warunek przechwycenia:} $v_p > v_t$ (ścigający szybszy od celu)
\end{frame}

\small

\begin{frame}{Stały kąt namiarowania (Constant Bearing)}
Ścigający porusza się pod \textbf{stałym kątem} $\beta$ względem kierunku do celu.

\vspace{0.4cm}
\textbf{Równania:}
\begin{align*}
\theta_{\text{target}} &= \arctan2(t_y - p_y, t_x - p_x) \\
\theta_{\text{pursuer}} &= \theta_{\text{target}} + \beta
\end{align*}

\vspace{0.4cm}
Trajektoria tworzy \textbf{spiralę logarytmiczną}.

\end{frame}

\begin{frame}{Nawigacja proporcjonalna (Proportional Navigation)}
Strategia szeroko stosowana w \textbf{naprowadzaniu pocisków}.

\vspace{0.3cm}
\textbf{Kąt LOS (Line of Sight):}
\begin{equation*}
\lambda = \arctan2(t_y - p_y, t_x - p_x)
\end{equation*}

\textbf{Zmiana kąta LOS:}
\begin{equation*}
\Delta\lambda = \lambda(t) - \lambda(t - \Delta t)
\end{equation*}

\textbf{Aktualizacja kierunku ścigającego:}
\begin{equation*}
\theta_{\text{new}} = \theta_{\text{current}} + N \cdot \Delta\lambda
\end{equation*}

gdzie $N$ - stała nawigacji (typowo 3-5). \textbf{Optymalna wartość:} $N \in [3, 4]$.
\end{frame}

\begin{frame}{Pościg cykliczny (Cyclic Pursuit)}
$n$ agentów na okręgu o promieniu $R$, każdy ściga kolejnego: $\vec{p}_{n+1} = \vec{p}_1$

\vspace{0.2cm}
\textbf{Równanie ruchu agenta $i$:}
\begin{equation*}
\frac{d\vec{p}_i}{dt} = v \cdot \frac{\vec{p}_{i+1} - \vec{p}_i}{\|\vec{p}_{i+1} - \vec{p}_i\|}
\end{equation*}

\vspace{0.2cm}
\textbf{Wyprowadzenie czasu zbieżności:}
\begin{enumerate}
    \item Kąt środkowy między sąsiednimi agentami: $\alpha = \frac{2\pi}{n}$
    \item Kierunek ruchu agenta $i$ tworzy z promieniem kąt: $\theta = \frac{\pi}{2} - \frac{\alpha}{2} = \frac{\pi}{2} - \frac{\pi}{n}$
    \item Składowa prędkości ku centrum: $v_r = v \cdot \cos\theta = v \cdot \sin\left(\frac{\pi}{n}\right)$
    \item Czas zbieżności: $T = \frac{R}{v_r} = \frac{R}{v \cdot \sin(\pi/n)}$
\end{enumerate}
\end{frame}

\begin{frame}{Pościg na sferze $S^2$ (Direct Pursuit)}
\small
Operujemy we \textbf{współrzędnych sferycznych} $(\theta, \phi)$, nie kartezjańskich:
\begin{itemize}
    \item $\theta \in [0, \pi]$ - kąt polarny (od bieguna północnego)
    \item $\phi \in [0, 2\pi)$ - kąt azymutalny (długość geograficzna)
\end{itemize}

\vspace{0.2cm}
\textbf{Metryka riemannowska:}
\begin{equation*}
ds^2 = R^2 d\theta^2 + R^2 \sin^2\theta \, d\phi^2
\end{equation*}

\vspace{0.3cm}
\textbf{Odległość kątowa:}
\begin{equation*}
d_{\text{angular}} = \arccos(\sin\theta_p \sin\theta_t + \cos\theta_p \cos\theta_t \cos(\phi_t - \phi_p))
\end{equation*}

\begin{frame}{Pościg na torusie (Direct Pursuit)}

\begin{frame}{Podejście ciągłe vs. dyskretne}
\textbf{Ciągłe} (\texttt{scipy.integrate.solve\_ivp}):
\begin{itemize}
    \item Adaptacyjny krok czasowy
    \item Kontrola błędu (tolerancje względne i bezwzględne)
    \item Detekcja zdarzeń (przechwycenie celu)
\end{itemize}

\vspace{0.4cm}
\textbf{Dyskretne} (iteracja z krokiem $\Delta t$):
\begin{equation*}
\vec{p}_{k+1} = \vec{p}_k + \Delta t \cdot \vec{v}_p(\vec{p}_k, \vec{t}_k)
\end{equation*}

\vspace{0.3cm}
Im mniejszy $\Delta t$, tym trajektoria bliższa rozwiązaniu ciągłemu.
\end{frame}

\section{Wnioski}
% ============================================

\begin{frame}{Podsumowanie wyników}
\textbf{Porównanie strategii:}
\begin{itemize}
    \item \textbf{Direct Pursuit} - najkrótsze trajektorie, prosty w implementacji
    \item \textbf{Proportional Navigation} ($N \in [3,4]$) - najlepszy kompromis efektywność/stabilność
    \item \textbf{Constant Bearing} - spirale logarytmiczne, najmniej efektywny
\end{itemize}

\vspace{0.4cm}
\textbf{Wymiar przestrzeni:}
\begin{itemize}
    \item Czas przechwycenia \textbf{niezależny od wymiaru} (dla ruchu jednoosiowego)
    \item Zweryfikowano numerycznie dla $n \leq 20$
\end{itemize}

\vspace{0.4cm}
\textbf{Geometria riemannowska (sfera $S^2$, torus):}
\begin{itemize}
    \item Pościg we współrzędnych krzywoliniowych $(\theta, \phi)$ i $(u, v)$
    \item Metryka definiuje odległość geodezyjną i kierunek pościgu
    \item Na torusie: nieliniowa zależność czasu od prędkości ścigającego
\end{itemize}
\end{frame}
